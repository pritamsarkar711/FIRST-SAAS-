import { NextRequest, NextResponse } from "next/server";
import { z } from "zod";
import { db } from "@/lib/db";
import { requireAdmin } from "@/lib/admin-guard";
import { logActivity } from "@/lib/activity";

export async function GET(req: NextRequest) {
  const { error } = await requireAdmin();
  if (error) return error;
  const q = req.nextUrl.searchParams.get("q")?.trim().toLowerCase() || "";
  const users = await db.user.findMany({
    where: q ? { OR: [{ email: { contains: q } }, { name: { contains: q } }] } : undefined,
    orderBy: { createdAt: "desc" },
    take: 100,
    select: {
      id: true, email: true, name: true, role: true, status: true, credits: true,
      referralCode: true, createdAt: true, twoFactorEnabled: true,
      _count: { select: { audits: true, payments: true } },
    },
  });
  return NextResponse.json({ users });
}

const actionSchema = z.object({
  id: z.string().min(1),
  action: z.enum(["suspend", "activate", "addCredits", "setCredits", "makeAdmin", "removeAdmin", "reset2fa", "delete"]),
  amount: z.number().int().optional(),
});

export async function POST(req: NextRequest) {
  const { user: admin, error } = await requireAdmin();
  if (error) return error;
  const parsed = actionSchema.safeParse(await req.json().catch(() => null));
  if (!parsed.success) return NextResponse.json({ error: "Invalid action." }, { status: 400 });
  const { id, action, amount } = parsed.data;

  const target = await db.user.findUnique({ where: { id } });
  if (!target) return NextResponse.json({ error: "User not found." }, { status: 404 });

  const log = (action: Parameters<typeof logActivity>[0]["action"], detail: string) =>
    logActivity({ action, actorEmail: admin.email, userId: target.id, detail });

  switch (action) {
    case "suspend":
      await db.user.update({ where: { id }, data: { status: "SUSPENDED" } });
      log("USER_SUSPEND", target.email);
      break;
    case "activate":
      await db.user.update({ where: { id }, data: { status: "ACTIVE" } });
      log("USER_UNSUSPEND", target.email);
      break;
    case "addCredits":
      if (!amount) return NextResponse.json({ error: "Amount required." }, { status: 400 });
      await db.user.update({ where: { id }, data: { credits: { increment: amount } } });
      log("USER_CREDITS_ADJUST", `${target.email} +${amount} credits`);
      break;
    case "setCredits":
      if (amount === undefined || amount < 0) return NextResponse.json({ error: "Amount required." }, { status: 400 });
      await db.user.update({ where: { id }, data: { credits: amount } });
      log("USER_CREDITS_ADJUST", `${target.email} set to ${amount} credits (was ${target.credits})`);
      break;
    case "makeAdmin":
      await db.user.update({ where: { id }, data: { role: "ADMIN" } });
      log("USER_ROLE_CHANGE", `${target.email} promoted to admin`);
      break;
    case "removeAdmin":
      if (target.id === admin.id) return NextResponse.json({ error: "You cannot remove your own admin role." }, { status: 400 });
      await db.user.update({ where: { id }, data: { role: "USER" } });
      log("USER_ROLE_CHANGE", `${target.email} demoted to user`);
      break;
    case "reset2fa":
      await db.user.update({ where: { id }, data: { twoFactorEnabled: false, twoFactorSecret: null } });
      log("AUTH_2FA_DISABLE", `Admin reset 2FA for ${target.email}`);
      break;
    case "delete":
      if (target.id === admin.id) return NextResponse.json({ error: "You cannot delete your own account." }, { status: 400 });
      await db.user.delete({ where: { id } });
      log("USER_DELETE", target.email);
      break;
  }
  return NextResponse.json({ ok: true });
}

  id: z.string().min(1),
  action: z.enum(["approve", "reject"]),
  note: z.string().max(300).optional(),
});

export async function POST(req: NextRequest) {
  const { user: admin, error } = await requireAdmin();
  if (error) return error;
  const parsed = actionSchema.safeParse(await req.json().catch(() => null));
  if (!parsed.success) return NextResponse.json({ error: "Invalid action." }, { status: 400 });

  const payment = await db.payment.findUnique({ where: { id: parsed.data.id }, select: { id: true, status: true, planName: true, amount: true, gateway: true, userId: true } });

  await notify(
    payment.userId,
    "PAYMENT",
    "Payment could not be verified",
    `Your ${payment.planName} order was rejected. ${parsed.data.note ? "Reason: " + parsed.data.note : "Contact support if you believe this is a mistake."}`,
    "/dashboard/billing"
  );
  logActivity({
    action: "PAYMENT_REJECT",
    actorEmail: admin.email,
    userId: payment.userId,
    detail: `${payment.planName} · $${payment.amount.toFixed(2)} · ${payment.gateway}${parsed.data.note ? " · " + parsed.data.note : ""}`,
  });
  return NextResponse.json({ ok: true, message: "Payment rejected." });

const schema = z.object({
  action: z.enum(["setRate", "withdrawal"]),
  rate: z.number().min(0).max(90).optional(),
  id: z.string().optional(),
  decision: z.enum(["approve", "reject", "complete"]).optional(),
});

export async function POST(req: NextRequest) {
  const { user: admin, error } = await requireAdmin();
  if (error) return error;
  const parsed = schema.safeParse(await req.json().catch(() => null));
  if (!parsed.success) return NextResponse.json({ error: "Invalid action." }, { status: 400 });

  if (parsed.data.action === "setRate" && parsed.data.rate !== undefined) {
    const { siteSettings } = await import("@/lib/settings");
    const site = await siteSettings();
    await import("@/lib/settings").then((m) => m.setSetting("site", { ...site, affiliateRate: parsed.data.rate }));

      actorEmail: admin.email,

    });
    return NextResponse.json({ ok: true });
  }

  if (parsed.data.action === "withdrawal" && parsed.data.id && parsed.data.decision) {
    const w = await db.withdrawal.findUnique({ where: { id: parsed.data.id } });
    if (!w) return NextResponse.json({ error: "Withdrawal not found." }, { status: 404 });
    const decision = parsed.data.decision;
    const log = (action: Parameters<typeof logActivity>[0]["action"], title: string) =>
      logActivity({ action, actorEmail: admin.email, userId: w.userId, detail: `$${w.amount.toFixed(2)} via ${w.method} to ${w.account} · ${title}` });
    if (decision === "reject") {
      // Return the money to cleared balance
      await db.affiliateEarning.create({
        data: { userId: w.userId, fromUserId: w.userId, amount: w.amount, status: "CLEARED" },
      });
      await db.withdrawal.update({ where: { id: w.id }, data: { status: "REJECTED", reviewedAt: new Date() } });
      await notify(w.userId, "WITHDRAWAL", "Withdrawal rejected", `Your $${w.amount.toFixed(2)} withdrawal was rejected and the amount was returned to your cleared balance.`, "/dashboard/affiliate");

    } else if (parsed.data.decision === "approve") {
      await db.withdrawal.update({ where: { id: w.id }, data: { status: "PROCESSING", reviewedAt: new Date() } });
      await notify(w.userId, "WITHDRAWAL", "Withdrawal processing", `Your $${w.amount.toFixed(2)} withdrawal via ${w.method} is being processed. Payouts arrive within 3 business days.`, "/dashboard/affiliate");
      log("WITHDRAW_APPROVE", "approved");
    } else {
      await db.withdrawal.update({ where: { id: w.id }, data: { status: "COMPLETED", reviewedAt: new Date() } });
      await notify(w.userId, "WITHDRAWAL", "Withdrawal paid", `Your $${w.amount.toFixed(2)} withdrawal via ${w.method} has been sent to ${w.account}.`, "/dashboard/affiliate");
      log("WITHDRAW_COMPLETE", "completed");
    }
    return NextResponse.json({ ok: true });
  }

  return NextResponse.json({ error: "Invalid action." }, { status: 400 });

const schema = z.object({
  id: z.string().min(1),
  action: z.enum(["reply", "close"]),
  reply: z.string().max(4000).optional(),
});

export async function POST(req: NextRequest) {
  const { error } = await requireAdmin();
  if (error) return error;
  const parsed = schema.safeParse(await req.json().catch(() => null));
  if (!parsed.success) return NextResponse.json({ error: "Invalid action." }, { status: 400 });

  const msg = await db.contactMessage.findUnique({ where: { id: parsed.data.id } });
  if (!msg) return NextResponse.json({ error: "Message not found." }, { status: 404 });

  if (parsed.data.action === "reply") {
    await db.contactMessage.update({
      where: { id: msg.id },
      data: { reply: parsed.data.reply || "", status: "REPLIED", repliedAt: new Date() },
    });
    if (msg.userId) {
      await notify(msg.userId, "MESSAGE", "Support replied", `We replied to your "${msg.reason}" message. Check the answer from support.`, "/contact");
    }
    // Email the reply if SMTP is configured (gracefully skipped otherwise)
    try {
      const { getSetting, SMTP_DEFAULTS } = await import("@/lib/settings");
      const smtp = await getSetting("smtp", SMTP_DEFAULTS);
      if (smtp.host && smtp.user && smtp.pass && parsed.data.reply) {
        const nodemailer = await import("nodemailer").catch(() => null);
        if (nodemailer) {
          const transport = nodemailer.createTransport({
            host: smtp.host, port: smtp.port, secure: smtp.port === 465,
            auth: { user: smtp.user, pass: smtp.pass },
          });
          await transport.sendMail({
            from: smtp.from || smtp.user,
            to: msg.email,
            subject: `Re: ${msg.reason} — Audit Form support`,
            text: parsed.data.reply,
          });
        }
      }
    } catch { /* SMTP errors are non-fatal; reply is stored in the panel */ }
    return NextResponse.json({ ok: true });
  }

  await db.contactMessage.update({ where: { id: msg.id }, data: { status: "CLOSED" } });
  return NextResponse.json({ ok: true });
}

  users: ["USER_CREDITS_ADJUST", "USER_SUSPEND", "USER_UNSUSPEND", "USER_DELETE", "USER_ROLE_CHANGE"],
  admin: ["PLAN_SAVE", "GATEWAY_SAVE", "SETTINGS_SAVE", "BLOG_PUBLISH", "ADMIN_EXPORT", "SMTP_TEST"],
};

export async function GET(req: NextRequest) {
  const { error } = await requireAdmin();
  if (error) return error;

  const group = req.nextUrl.searchParams.get("group") || "all";
  const q = req.nextUrl.searchParams.get("q")?.trim().toLowerCase() || "";
  const page = Math.max(1, parseInt(req.nextUrl.searchParams.get("page") || "1", 10) || 1);
  const perPage = 50;

  const actions = GROUPS[group];
  const where: Record<string, unknown> = {};
  if (actions) where.action = { in: actions };
  if (q) {
    where.OR = [
      { actorEmail: { contains: q } },
      { detail: { contains: q } },
      { ip: { contains: q } },
    ];
  }

  const [rows, total] = await Promise.all([
    db.activityLog.findMany({
      where,
      orderBy: { createdAt: "desc" },
      skip: (page - 1) * perPage,
      take: perPage,
    }),
    db.activityLog.count({ where }),
  ]);

  return NextResponse.json({
    rows,
    total,
    page,
    perPage,
    pages: Math.max(1, Math.ceil(total / perPage)),
  });
}

      </div>`,
