    61→      log("USER_CREDITS_ADJUST", `${target.email} set to ${amount} credits (was ${target.credits})`);

    64→      await db.user.update({ where: { id }, data: { role: "ADMIN" } });
    65→      log("USER_ROLE_CHANGE", `${target.email} promoted to admin`);
    66→      break;
    67→    case "removeAdmin":
    68→      if (target.id === admin.id) return NextResponse.json({ error: "You cannot remove your own admin role." }, { status: 400 });
    69→      await db.user.update({ where: { id }, data: { role: "USER" } });
    70→      log("USER_ROLE_CHANGE", `${target.email} demoted to user`);

    74→      log("AUTH_2FA_DISABLE", `Admin reset 2FA for ${target.email}`);
    75→      break;
    76→    case "delete":
    77→      if (target.id === admin.id) return NextResponse.json({ error: "You cannot delete your own account." }, { status: 400 });
    78→      await db.user.delete({ where: { id } });
    79→      log("USER_DELETE", target.email);
    80→      break;
    81→  }
    82→  return NextResponse.json({ ok: true });
    83→}
